import pandas as pd

# 从 CSV 加载股票池
df_codes = pd.read_csv("sorted_high_preclose_ratio_2020.csv")
market_list = df_codes["code"].drop_duplicates().tolist()
print(f"股票池加载完成，数量: {len(market_list)}")

# 格式统一
market_list = [
    code.upper().replace("SZ.", "SZ").replace("SH.", "SH")
    for code in market_list
]

# 检查时间段
start_time = "2020-09-14 09:30:00"
end_time = "2021-01-18 14:59:00"
fields = ["$close"]

no_data_list = []
filtered_market = []

print(f"总共 {len(market_list)} 支股票，开始检查分钟数据...")

for code in market_list:
    try:
        data = D.features([code], fields, start_time=start_time, end_time=end_time, freq="1min")
        if data.shape[0] == 0:
            no_data_list.append(code)
            print(f"⚠️ {code} 没有数据")
        else:
            filtered_market.append(code)
            print(f"✅ {code} 有数据，数据量：{data.shape[0]}")
    except Exception as e:
        no_data_list.append(code)
        print(f"❌ {code} 检查出错: {e}")

print("\n检查完成")
if no_data_list:
    print(f"以下股票没有分钟数据：")
    print(no_data_list)

print(f"筛选后股票池数量: {len(filtered_market)}")
